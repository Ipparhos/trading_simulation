```python
    idx: int,
    net_pnl: float,
    base_time: datetime,
    hours_apart: int = 8,

    entry_t = base_time + timedelta(hours=idx * hours_apart)

        symbol="BTC/USD",
```

```python
from datetime import datetime, timedelta, timezone
from pathlib import Path
from typing import Any

import pandas as pd
import pytest

from src.ingestion.kraken_loader import (
    KrakenDataLoader,
    df_to_bars,
    timeframe_to_timedelta,
    validate_ohlcv_df,
)


class FakeExchange:
    """Deterministic in-memory CCXT stub for testing pagination and caching."""

    def __init__(self, rows: list[list[Any]]) -> None:
        self.rows = rows
        self.call_count = 0

    def fetch_ohlcv(
        self, symbol: str, timeframe: str = "15m", since: int = 0, limit: int = 720
    ) -> list[list[Any]]:
        del symbol, timeframe
        self.call_count += 1
        eligible = [r for r in self.rows if int(r[0]) >= since]
        return eligible[:limit]


def _make_15m_rows(start: datetime, count: int) -> list[list[Any]]:
    rows: list[list[Any]] = []
    delta = timedelta(minutes=15)
    for i in range(count):
        ts_ms = int((start + i * delta).timestamp() * 1000)
        price = 100_000.0 + i * 10.0
        rows.append([ts_ms, price, price + 20.0, price - 20.0, price + 5.0, 1.5])
    return rows


def test_parquet_caching_and_pagination(tmp_path: Path) -> None:
    """Verify loader paginates across batches, writes Parquet, and serves cache hits."""
    start = datetime(2026, 1, 1, 0, 0, tzinfo=timezone.utc)
    until = start + timedelta(minutes=15 * 5)
    rows = _make_15m_rows(start, 6)

    fake_ex = FakeExchange(rows)
    loader = KrakenDataLoader(cache_dir=tmp_path, exchange=fake_ex, page_limit=2)

    df = loader.load_ohlcv("BTC/USD", "15m", since=start, until=until, strict_gaps=True)
    assert len(df) == 6
    assert fake_ex.call_count == 3
    assert loader.cache_path("BTC/USD", "15m").exists()

    bars = df_to_bars(df)
    assert len(bars) == 6
    assert bars[0].timestamp == start
    assert bars[-1].timestamp == until

    # Second load over covered range should hit Parquet cache without calling exchange.
    df_cached = loader.load_ohlcv("BTC/USD", "15m", since=start, until=until)
    assert len(df_cached) == 6
    assert fake_ex.call_count == 3
```

````markdown
# ARCHITECTURE.md — Kraken Prop & Funded Scenario Simulation Engine

## 1. System Objective & Core Philosophy
You are tasked with building a high-precision, event-driven quantitative simulation and Monte Carlo engine in Python (`3.12+`) designed to stress-test trading strategies against **Kraken Prop (Kraken Pro)** and **Kraken Funded** evaluation rules.

### Non-Negotiable Engineering Constraints (Do Not Violate)
1. **Zero Lookahead Bias:** Strategies may only access closed historical bars up to index `i`. Orders generated at the close of bar `i` execute at the open of bar `i+1` (market) or intrabar during `i+1..n` (limit/stop).
2. **Pessimistic Intrabar Wick Resolution:** Never evaluate Drawdown or Stop-Loss/Take-Profit solely on candle closes. During any open position on bar `i`, the engine MUST evaluate the bar's `Low` (for Longs) and `High` (for Shorts) against both the position Stop-Loss and the account's real-time Equity Breach Floor. If a candle's `High` and `Low` touch both the Stop-Loss and Take-Profit within the same 1m/5m bar, **the engine must assume the Stop-Loss was hit first (Pessimistic Execution).**
3. **Exact Fee & Rollover State Machine:** Fees reduce account `balance` immediately and count directly against both Daily Loss and Maximum Drawdown limits.
4. **Vectorized Signal Prep, Sequential State Machine:** Indicators may be pre-computed vectorially, but account equity, 00:30 UTC daily resets, 4-hour swap rollovers, and breach checks MUST run sequentially bar-by-bar.
````

```python
from datetime import datetime, timedelta, timezone
from pathlib import Path
from typing import Any, Optional

import ccxt
import pandas as pd

from src.schemas.orders import ensure_utc_timestamp
from src.schemas.telemetry import OhlcBar

DEFAULT_CACHE_DIR = Path(__file__).resolve().parents[2] / "data" / "parquet"
REQUIRED_OHLCV_COLUMNS = ["timestamp", "open", "high", "low", "close", "volume"]

SUPPORTED_TIMEFRAMES: dict[str, timedelta] = {
    "1m": timedelta(minutes=1),
    "5m": timedelta(minutes=5),
    "15m": timedelta(minutes=15),
    "1h": timedelta(hours=1),
    "4h": timedelta(hours=4),
    "1d": timedelta(days=1),
}


def timeframe_to_timedelta(timeframe: str) -> timedelta:
    """Map a CCXT timeframe string to a Python timedelta."""
    if timeframe not in SUPPORTED_TIMEFRAMES:
        raise ValueError(f"Unsupported timeframe: '{timeframe}'")
    return SUPPORTED_TIMEFRAMES[timeframe]


def validate_ohlcv_df(
    df: pd.DataFrame,
    timeframe: str,
    strict_gaps: bool = False,
) -> list[datetime]:
    """Validate OHLCV columns, UTC awareness, ordering, uniqueness, and bar gaps."""
    missing_cols = [c for c in REQUIRED_OHLCV_COLUMNS if c not in df.columns]
    if missing_cols:
        raise ValueError(f"Missing required OHLCV columns: {missing_cols}")
    if df.empty:
        return []

    ts_dtype = df["timestamp"].dtype
    if not isinstance(ts_dtype, pd.DatetimeTZDtype) or str(ts_dtype.tz) != "UTC":
        raise ValueError("DataFrame 'timestamp' column must be timezone-aware UTC.")

    if df["timestamp"].duplicated().any():
        raise ValueError("DataFrame contains duplicate timestamps.")
    if not df["timestamp"].is_monotonic_increasing:
        raise ValueError("DataFrame timestamps must be sorted in ascending order.")

    expected_delta = timeframe_to_timedelta(timeframe)
    diffs = df["timestamp"].diff()
    gap_indices = df.index[diffs > expected_delta].tolist()

    gaps: list[datetime] = []
    for idx in gap_indices:
        pos = df.index.get_loc(idx)
        prev_ts = df["timestamp"].iloc[pos - 1].to_pydatetime()
        gaps.append(prev_ts + expected_delta)

    if strict_gaps and gaps:
        raise ValueError(
            f"Detected {len(gaps)} missing OHLCV interval(s); first gap at {gaps[0].isoformat()}"
        )
    return gaps


def df_to_bars(df: pd.DataFrame) -> list[OhlcBar]:
    """Convert a validated OHLCV DataFrame into a sequence of OhlcBar objects."""
    bars: list[OhlcBar] = []
    for row in df.itertuples(index=False):
        ts = row.timestamp.to_pydatetime()
        bars.append(
            OhlcBar(
                timestamp=ts,
                open=float(row.open),
                high=float(row.high),
                low=float(row.low),
                close=float(row.close),
                volume=float(row.volume),
            )
        )
    return bars


class KrakenDataLoader:
    """CCXT historical OHLCV fetcher with local Parquet caching and gap validation."""

    def __init__(
        self,
        cache_dir: Path = DEFAULT_CACHE_DIR,
        exchange: Optional[Any] = None,
        page_limit: int = 720,
    ) -> None:
        self.cache_dir = cache_dir
        self._exchange = exchange
        self.page_limit = page_limit

    @property
    def exchange(self) -> Any:
        if self._exchange is None:
            self._exchange = ccxt.kraken({"enableRateLimit": True})
        return self._exchange

    def cache_path(self, symbol: str, timeframe: str) -> Path:
        slug = symbol.lower().replace("/", "_").replace("-", "_")
        return self.cache_dir / f"{slug}_{timeframe}.parquet"

    def _fetch_range(
        self, symbol: str, timeframe: str, since_utc: datetime, until_utc: datetime
    ) -> pd.DataFrame:
        delta = timeframe_to_timedelta(timeframe)
        step_ms = int(delta.total_seconds() * 1000)
        cursor_ms = int(since_utc.timestamp() * 1000)
        until_ms = int(until_utc.timestamp() * 1000)

        rows: list[list[Any]] = []
        while cursor_ms <= until_ms:
            batch = self.exchange.fetch_ohlcv(
                symbol, timeframe=timeframe, since=cursor_ms, limit=self.page_limit
            )
            if not batch:
                break
            rows.extend(batch)
            last_ms = int(batch[-1][0])
            next_ms = last_ms + step_ms
            if next_ms <= cursor_ms or (
                len(batch) < self.page_limit and last_ms >= until_ms
            ):
                break
            cursor_ms = next_ms

        if not rows:
            return pd.DataFrame(
                {
                    "timestamp": pd.Series(dtype="datetime64[ns, UTC]"),
                    "open": pd.Series(dtype="float64"),
                    "high": pd.Series(dtype="float64"),
                    "low": pd.Series(dtype="float64"),
                    "close": pd.Series(dtype="float64"),
                    "volume": pd.Series(dtype="float64"),
                }
            )

        raw_df = pd.DataFrame(rows, columns=REQUIRED_OHLCV_COLUMNS)
        raw_df["timestamp"] = pd.to_datetime(raw_df["timestamp"], unit="ms", utc=True)
        for col in ("open", "high", "low", "close", "volume"):
            raw_df[col] = raw_df[col].astype(float)
        return raw_df

    def load_ohlcv(
        self,
        symbol: str,
        timeframe: str,
        since: datetime,
        until: Optional[datetime] = None,
        force_refresh: bool = False,
        strict_gaps: bool = False,
    ) -> pd.DataFrame:
        """Load OHLCV bars from Parquet cache or CCXT Kraken, returning validated UTC DataFrame."""
        since_utc = ensure_utc_timestamp(since)
        until_utc = (
            ensure_utc_timestamp(until)
            if until is not None
            else datetime.now(timezone.utc)
        )
        if until_utc < since_utc:
            raise ValueError("'until' timestamp must be >= 'since' timestamp.")

        delta = timeframe_to_timedelta(timeframe)
        path = self.cache_path(symbol, timeframe)
        cached_df: Optional[pd.DataFrame] = None

        if path.exists() and not force_refresh:
            cached_df = pd.read_parquet(path)
            if not cached_df.empty:
                cached_min = cached_df["timestamp"].min().to_pydatetime()
                cached_max = cached_df["timestamp"].max().to_pydatetime()
                if cached_min <= since_utc and cached_max >= until_utc - delta:
                    mask = (cached_df["timestamp"] >= since_utc) & (
                        cached_df["timestamp"] <= until_utc
                    )
                    result_df = cached_df.loc[mask].reset_index(drop=True)
                    validate_ohlcv_df(result_df, timeframe, strict_gaps=strict_gaps)
                    return result_df

        fetched_df = self._fetch_range(symbol, timeframe, since_utc, until_utc)
        if cached_df is not None and not cached_df.empty:
            combined = pd.concat([cached_df, fetched_df], ignore_index=True)
        else:
            combined = fetched_df

        if not combined.empty:
            combined = (
                combined.drop_duplicates(subset=["timestamp"], keep="last")
                .sort_values("timestamp")
                .reset_index(drop=True)
            )
            path.parent.mkdir(parents=True, exist_ok=True)
            combined.to_parquet(path, index=False)
            mask = (combined["timestamp"] >= since_utc) & (
                combined["timestamp"] <= until_utc
            )
            result_df = combined.loc[mask].reset_index(drop=True)
        else:
            result_df = combined

        validate_ohlcv_df(result_df, timeframe, strict_gaps=strict_gaps)
        return result_df
```

```python
            spread_markup_bps_per_side=self.config.spread_markup_bps_per_side,
            slippage_bps=slippage_bps,
        )
        entry_fee = compute_commission(
            notional=qty * fill_price,
            commission_bps_per_side=self.config.commission_bps_per_side,
        )
        balance_after = guard.record_balance_delta(-entry_fee, update_hwm=False)
        ledger.append(
            LedgerEntry(
                timestamp=entry_time_utc,
                entry_type=LedgerEntryType.ENTRY_FEE,
                amount=-entry_fee,
                balance_after=balance_after,
            )
        )
        return Position(
            symbol=symbol,
            side=side,
            qty=qty,
            entry_price=fill_price,
            entry_time=entry_time_utc,
            stop_loss=stop_loss,
            take_profit=take_profit,
            realized_entry_fee=entry_fee,
            accumulated_swap_fees=0.0,
        )

    def close_position(
        self,
        guard: RulesGuard,
        ledger: list[LedgerEntry],
        trades: list[TradeRecord],
        position: Position,
        raw_price: float,
        exit_time: datetime,
        reason: ExitReason,
        slippage_bps: float = 0.0,
    ) -> TradeRecord:
        """Close an open position, update closed balance, and record ledger/trade entries."""
        exit_time_utc = ensure_utc_timestamp(exit_time)
        exit_fill = effective_fill_price(
            raw_price=raw_price,
            side=position.side,
            is_entry=False,
            spread_markup_bps_per_side=self.config.spread_markup_bps_per_side,
            slippage_bps=slippage_bps,
        )
        if position.side == Side.LONG:
            gross_pnl = position.qty * (exit_fill - position.entry_price)
        else:
            gross_pnl = position.qty * (position.entry_price - exit_fill)

        exit_fee = compute_commission(
            notional=position.qty * exit_fill,
            commission_bps_per_side=self.config.commission_bps_per_side,
        )

        balance_after_pnl = guard.record_balance_delta(gross_pnl, update_hwm=False)
        ledger.append(
            LedgerEntry(
                timestamp=exit_time_utc,
                entry_type=LedgerEntryType.REALIZED_PNL,
                amount=gross_pnl,
                balance_after=balance_after_pnl,
            )
        )
        balance_after_fee = guard.record_balance_delta(-exit_fee, update_hwm=True)
        ledger.append(
            LedgerEntry(
                timestamp=exit_time_utc,
                entry_type=LedgerEntryType.EXIT_FEE,
                amount=-exit_fee,
                balance_after=balance_after_fee,
            )
        )

        net_pnl = (
            gross_pnl
            - position.realized_entry_fee
            - exit_fee
            - position.accumulated_swap_fees
        )
        trade = TradeRecord(
            symbol=position.symbol,
            side=position.side,
            qty=position.qty,
            entry_price=position.entry_price,
            exit_price=exit_fill,
            entry_time=position.entry_time,
            exit_time=exit_time_utc,
            exit_reason=reason,
            gross_pnl=gross_pnl,
            entry_fee=position.realized_entry_fee,
            exit_fee=exit_fee,
            swap_fees=position.accumulated_swap_fees,
            net_pnl=net_pnl,
        )
        trades.append(trade)
        return trade

    @staticmethod
    def adverse_raw_price(
        position: Position,
        open_price: float,
        high_price: float,
        low_price: float,
    ) -> tuple[float, bool]:
        """Return (adverse_raw_price, stop_loss_touched) for the current bar."""
        if position.side == Side.LONG:
            if position.stop_loss is not None and low_price <= position.stop_loss:
                if open_price <= position.stop_loss:
                    return open_price, True
                return max(low_price, position.stop_loss), True
            return low_price, False

        if position.stop_loss is not None and high_price >= position.stop_loss:
            if open_price >= position.stop_loss:
                return open_price, True
            return min(high_price, position.stop_loss), True
        return high_price, False

    @staticmethod
    def favorable_raw_price(
        position: Position,
        open_price: float,
        high_price: float,
        low_price: float,
    ) -> tuple[float, bool]:
        """Return (favorable_raw_price, take_profit_touched) for the current bar."""
        del open_price
        if position.side == Side.LONG:
            tp_hit = (
                position.take_profit is not None and high_price >= position.take_profit
            )
            return high_price, tp_hit

        tp_hit = (
            position.take_profit is not None and low_price <= position.take_profit
        )
        return low_price, tp_hit

    def solve_target_raw_price(
        self,
        position: Position,
        guard: RulesGuard,
        slippage_bps: float = 0.0,
    ) -> float:
        """Solve for the exact raw market price where live_equity hits profit_target_equity."""
        comm_rate = self.config.commission_bps_per_side / BPS_DIVISOR
        spread_rate = (
            self.config.spread_markup_bps_per_side + slippage_bps
        ) / BPS_DIVISOR
        target = guard.profit_target_equity
        balance = guard.closed_balance

        if position.side == Side.LONG:
            effective_exit = (
                target - balance + position.qty * position.entry_price
            ) / (position.qty * (1.0 - comm_rate))
            return effective_exit / (1.0 - spread_rate)

        effective_exit = (
            balance + position.qty * position.entry_price - target
        ) / (position.qty * (1.0 + comm_rate))
        return effective_exit / (1.0 + spread_rate)
```

```python
from src.schemas.orders import Side

BPS_DIVISOR = 10_000.0
SWAP_INTERVALS_PER_DAY = 6.0


def effective_fill_price(
    raw_price: float,
    side: Side,
    is_entry: bool,
    spread_markup_bps_per_side: float,
    slippage_bps: float = 0.0,
) -> float:
    """Compute effective execution price after spread markup and slippage."""
    total_adjustment = (spread_markup_bps_per_side + slippage_bps) / BPS_DIVISOR
    is_buying = (side == Side.LONG and is_entry) or (
        side == Side.SHORT and not is_entry
    )
    if is_buying:
        return raw_price * (1.0 + total_adjustment)
    return raw_price * (1.0 - total_adjustment)


def compute_commission(notional: float, commission_bps_per_side: float) -> float:
    """Compute trading commission in USD for a given fill notional."""
    return notional * (commission_bps_per_side / BPS_DIVISOR)


def compute_4h_swap(notional: float, daily_swap_rate_pct: float) -> float:
    """Compute a single 4-hour rollover swap fee (one-sixth of the daily rate)."""
    return notional * (daily_swap_rate_pct / 100.0 / SWAP_INTERVALS_PER_DAY)
```

```python
from typing import Optional

from src.engine.fee_calculator import compute_commission, effective_fill_price
from src.schemas.config import ChallengeConfig, DrawdownType, MdlResetBasis
from src.schemas.orders import Position, Side
from src.schemas.telemetry import AccountStatus


class RulesGuard:
    """Real-time balance, live equity, drawdown floor, and profit target guard."""

    def __init__(self, config: ChallengeConfig) -> None:
        self.config = config
        self.closed_balance: float = config.initial_balance
        self.highest_closed_balance: float = config.initial_balance
        self.active_mdd_floor: float = config.initial_balance * (
            1.0 - config.max_drawdown_pct
        )
        self.active_mdl_floor: Optional[float] = (
            config.initial_balance * (1.0 - config.max_daily_loss_pct)
            if config.max_daily_loss_pct is not None
            else None
        )
        self.profit_target_equity: float = config.initial_balance * (
            1.0 + config.profit_target_pct
        )
        self.status: AccountStatus = AccountStatus.ACTIVE

    def record_balance_delta(self, delta: float, update_hwm: bool = True) -> float:
        """Apply a realized balance change and optionally update closed high-water mark."""
        self.closed_balance += delta
        if update_hwm and self.closed_balance > self.highest_closed_balance:
            self.highest_closed_balance = self.closed_balance
        return self.closed_balance

    def live_equity(
        self,
        position: Optional[Position],
        raw_market_price: float,
        slippage_bps: float = 0.0,
    ) -> float:
        """Compute mark-to-market equity net of estimated close spread and commission."""
        if position is None:
            return self.closed_balance

        effective_exit = effective_fill_price(
            raw_price=raw_market_price,
            side=position.side,
            is_entry=False,
            spread_markup_bps_per_side=self.config.spread_markup_bps_per_side,
            slippage_bps=slippage_bps,
        )
        if position.side == Side.LONG:
            unrealized_pnl = position.qty * (effective_exit - position.entry_price)
        else:
            unrealized_pnl = position.qty * (position.entry_price - effective_exit)

        estimated_exit_comm = compute_commission(
            notional=position.qty * effective_exit,
            commission_bps_per_side=self.config.commission_bps_per_side,
        )
        return self.closed_balance + unrealized_pnl - estimated_exit_comm

    def on_daily_reset(self, current_close_equity: float) -> None:
        """Recalculate active MDL floor and trailing-closed MDD floor at daily reset."""
        if self.config.max_daily_loss_pct is not None:
            if self.config.mdl_reset_basis == MdlResetBasis.BALANCE:
                basis = self.closed_balance
            elif self.config.mdl_reset_basis == MdlResetBasis.EQUITY:
                basis = current_close_equity
            else:
                basis = max(self.closed_balance, current_close_equity)
            self.active_mdl_floor = basis * (1.0 - self.config.max_daily_loss_pct)

        if self.config.drawdown_type == DrawdownType.TRAILING_CLOSED:
            max_dd_dollars = self.config.initial_balance * self.config.max_drawdown_pct
            self.active_mdd_floor = self.highest_closed_balance - max_dd_dollars

    def check_breach(self, equity: float) -> Optional[AccountStatus]:
        """Pure query: return breach status if equity hits MDD or MDL floor."""
        if equity <= self.active_mdd_floor:
            return AccountStatus.BREACHED_MDD
        if self.active_mdl_floor is not None and equity <= self.active_mdl_floor:
            return AccountStatus.BREACHED_MDL
        return None

    def check_target_reached(self, equity: float) -> bool:
        """Pure query: return True if equity meets or exceeds profit target."""
        return equity >= self.profit_target_equity

    def transition_status(self, status: AccountStatus) -> None:
        """Explicitly transition the account status after accounting is finalized."""
        self.status = status
```

```python
from collections.abc import Sequence
from datetime import datetime
from typing import Iterable, Optional, overload

from src.engine.clock import EngineClock, next_daily_reset_after
from src.engine.execution import ExecutionEngine
from src.engine.fee_calculator import compute_4h_swap
from src.engine.rules_guard import RulesGuard
from src.schemas.config import ChallengeConfig
from src.schemas.orders import (

    LedgerEntryType,
    OrderType,
    Position,
    Side,
    TradeRecord,
    ensure_utc_timestamp,

from src.schemas.telemetry import (

    OhlcBar,
    SimulationResult,

from src.strategies.base import BaseStrategy, CloseIntent, OrderIntent, StrategySignal


class _BoundedHistoryView(Sequence[OhlcBar]):
    """Read-only view of bars[0..visible_len-1] preventing lookahead access."""

    def __init__(self, source: Sequence[OhlcBar], visible_len: int) -> None:
        self._source = source
        self._visible_len = visible_len

    def __len__(self) -> int:
        return self._visible_len

    @overload
    def __getitem__(self, index: int) -> OhlcBar: ...

    @overload
    def __getitem__(self, index: slice) -> Sequence[OhlcBar]: ...

    def __getitem__(self, index: int | slice) -> OhlcBar | Sequence[OhlcBar]:
        if isinstance(index, slice):
            start, stop, step = index.indices(self._visible_len)
            return [self._source[i] for i in range(start, stop, step)]
        if index < 0:
            index += self._visible_len
        if index < 0 or index >= self._visible_len:
            raise IndexError("Lookahead violation or out-of-bounds history access.")
        return self._source[index]


class Simulator:
    """Sequential event-driven bar simulator enforcing Kraken Prop & Funded rules.

    Assumes sub-4-hour bar granularity (e.g., 1m/5m/15m) so 4h swap rollovers
    align with bar opens. Bars spanning multiple 4h boundaries value each rollover
    at that bar's ``open`` price.
    """

    def __init__(self, config: ChallengeConfig, slippage_bps: float = 0.0) -> None:
        self.config = config
        self.slippage_bps = slippage_bps
        self.clock = EngineClock(
            reset_hour=config.daily_reset_utc_hour,
            reset_minute=config.daily_reset_utc_minute,
        )
        self.guard = RulesGuard(config)
        self.execution = ExecutionEngine(config)
        self.position: Optional[Position] = None
        self.trades: list[TradeRecord] = []
        self.ledger: list[LedgerEntry] = []
        self.snapshots: list[BarSnapshot] = []
        self.breach_timestamp: Optional[datetime] = None
        self.breach_reason: Optional[str] = None

    def open_position(
        self,
        symbol: str,
        side: Side,
        qty: float,
        raw_price: float,
        entry_time: datetime,
        stop_loss: Optional[float] = None,
        take_profit: Optional[float] = None,
        slippage_bps: Optional[float] = None,
    ) -> Position:
        """Open a position at entry_time, synchronizing the engine clock if flat."""
        if self.guard.status != AccountStatus.ACTIVE:
            raise RuntimeError(
                f"Cannot open position when account status is {self.guard.status.value}."
            )
        if self.position is not None:
            raise RuntimeError("A position is already open.")

        entry_utc = ensure_utc_timestamp(entry_time)
        if self.clock.last_processed_time is None:
            self.clock.last_processed_time = entry_utc
        elif entry_utc < self.clock.last_processed_time:
            raise ValueError("Position entry_time precedes clock last_processed_time.")
        elif entry_utc > self.clock.last_processed_time:
            _, did_reset = self.clock.advance(entry_utc)
            if did_reset:
                self.guard.on_daily_reset(self.guard.closed_balance)

        effective_slippage = (
            self.slippage_bps if slippage_bps is None else slippage_bps
        )
        self.position = self.execution.open_position(
            guard=self.guard,
            ledger=self.ledger,
            symbol=symbol,
            side=side,
            qty=qty,
            raw_price=raw_price,
            entry_time=entry_utc,
            stop_loss=stop_loss,
            take_profit=take_profit,
            slippage_bps=effective_slippage,
        )
        return self.position

    def close_position_signal(
        self,
        raw_price: float,
        exit_time: datetime,
        slippage_bps: Optional[float] = None,
    ) -> Optional[TradeRecord]:
        """Close the active position manually due to a strategy signal."""
        if self.position is None:
            return None
        exit_utc = ensure_utc_timestamp(exit_time)
        effective_slippage = (
            self.slippage_bps if slippage_bps is None else slippage_bps
        )
        trade = self.execution.close_position(
            guard=self.guard,
            ledger=self.ledger,
            trades=self.trades,
            position=self.position,
            raw_price=raw_price,
            exit_time=exit_utc,
            reason=ExitReason.SIGNAL_EXIT,
            slippage_bps=effective_slippage,
        )
        self.position = None
        return trade

    def _force_liquidate_on_breach(
        self,
        raw_price: float,
        exit_time: datetime,
        breach_status: AccountStatus,
        reason_text: str,
        slippage_bps: float = 0.0,
    ) -> None:
        if self.position is not None:
            self.execution.close_position(
                guard=self.guard,
                ledger=self.ledger,
                trades=self.trades,
                position=self.position,
                raw_price=raw_price,
                exit_time=exit_time,
                reason=ExitReason.BREACH_LIQUIDATE,
                slippage_bps=slippage_bps,
            )
            self.position = None
        self.guard.transition_status(breach_status)
        self.breach_timestamp = exit_time
        self.breach_reason = reason_text

    def _record_snapshot(
        self,
        timestamp: datetime,
        worst_intrabar_equity: float,
        best_intrabar_equity: float,
        close_equity: float,
    ) -> BarSnapshot:
        snapshot = BarSnapshot(
            timestamp=timestamp,
            closed_balance=self.guard.closed_balance,
            worst_intrabar_equity=worst_intrabar_equity,
            best_intrabar_equity=best_intrabar_equity,
            close_equity=close_equity,
            active_mdd_floor=self.guard.active_mdd_floor,
            active_mdl_floor=self.guard.active_mdl_floor,
            status=self.guard.status,
        )
        self.snapshots.append(snapshot)
        return snapshot

    def _process_step_a_clock_events(self, bar: OhlcBar) -> Optional[BarSnapshot]:
        """Process 4h swap rollovers and daily 00:30 UTC reset in chronological order.

        Note: Swap notional is valued at ``bar.open`` for every 4h boundary crossed
        in ``(prev_time, bar.timestamp]``.
        """
        prev_time = self.clock.last_processed_time
        rollovers, did_reset = self.clock.advance(bar.timestamp)
        if prev_time is None:
            return None

        events: list[tuple[datetime, int]] = [(rt, 0) for rt in rollovers]
        if did_reset:
            reset_time = next_daily_reset_after(
                prev_time,
                reset_hour=self.clock.reset_hour,
                reset_minute=self.clock.reset_minute,
            )
            events.append((reset_time, 1))
        events.sort(key=lambda item: (item[0], item[1]))

        for event_time, event_kind in events:
            if event_kind == 0:
                if self.position is None:
                    continue
                notional = self.position.qty * bar.open
                swap_fee = compute_4h_swap(notional, self.config.daily_swap_rate_pct)
                balance_after = self.guard.record_balance_delta(
                    -swap_fee, update_hwm=False
                )
                self.position.accumulated_swap_fees += swap_fee
                self.ledger.append(
                    LedgerEntry(
                        timestamp=event_time,
                        entry_type=LedgerEntryType.SWAP_FEE,
                        amount=-swap_fee,
                        balance_after=balance_after,
                    )
                )
                open_equity = self.guard.live_equity(self.position, bar.open)
                breach = self.guard.check_breach(open_equity)
                if breach is not None:
                    self._force_liquidate_on_breach(
                        raw_price=bar.open,
                        exit_time=bar.timestamp,
                        breach_status=breach,
                        reason_text=f"Swap rollover breach at {event_time.isoformat()}",
                    )
                    return self._record_snapshot(
                        timestamp=bar.timestamp,
                        worst_intrabar_equity=open_equity,
                        best_intrabar_equity=open_equity,
                        close_equity=self.guard.closed_balance,
                    )
            else:
                open_equity = self.guard.live_equity(self.position, bar.open)
                self.guard.on_daily_reset(current_close_equity=open_equity)
        return None

    def _apply_pending_signal_at_open(
        self, bar: OhlcBar, signal: StrategySignal
    ) -> None:
        if isinstance(signal, CloseIntent):
            if self.position is not None:
                self.close_position_signal(
                    raw_price=bar.open, exit_time=bar.timestamp
                )
            return

        if isinstance(signal, OrderIntent) and self.position is None:
            if signal.order_type == OrderType.MARKET:
                self.open_position(
                    symbol=signal.symbol,
                    side=signal.side,
                    qty=signal.qty,
                    raw_price=bar.open,
                    entry_time=bar.timestamp,
                    stop_loss=signal.stop_loss,
                    take_profit=signal.take_profit,
                )
            elif (
                signal.order_type == OrderType.LIMIT
                and signal.limit_price is not None
            ):
                if signal.side == Side.LONG and bar.low <= signal.limit_price:
                    fill_raw = min(bar.open, signal.limit_price)
                    self.open_position(
                        symbol=signal.symbol,
                        side=signal.side,
                        qty=signal.qty,
                        raw_price=fill_raw,
                        entry_time=bar.timestamp,
                        stop_loss=signal.stop_loss,
                        take_profit=signal.take_profit,
                        slippage_bps=0.0,
                    )
                elif signal.side == Side.SHORT and bar.high >= signal.limit_price:
                    fill_raw = max(bar.open, signal.limit_price)
                    self.open_position(
                        symbol=signal.symbol,
                        side=signal.side,
                        qty=signal.qty,
                        raw_price=fill_raw,
                        entry_time=bar.timestamp,
                        stop_loss=signal.stop_loss,
                        take_profit=signal.take_profit,
                        slippage_bps=0.0,
                    )

    def process_bar(
        self, bar: OhlcBar, pending_signal: Optional[StrategySignal] = None
    ) -> Optional[BarSnapshot]:
        """Process a single OHLC bar strictly in order: Clock -> Adverse -> Favorable."""
        if self.guard.status != AccountStatus.ACTIVE:
            return None

        # Step A: Clock events (4h rollovers & daily 00:30 UTC reset)
        step_a_halt = self._process_step_a_clock_events(bar)
        if step_a_halt is not None:
            return step_a_halt

        if pending_signal is not None:
            self._apply_pending_signal_at_open(bar, pending_signal)

        if self.position is None:
            balance = self.guard.closed_balance
            if self.guard.check_target_reached(balance):
                self.guard.transition_status(AccountStatus.PASSED)
            return self._record_snapshot(
                timestamp=bar.timestamp,
                worst_intrabar_equity=balance,
                best_intrabar_equity=balance,
                close_equity=balance,
            )

        # Step B: Intrabar adverse path first (Pessimistic execution)
        adverse_price, sl_touched = self.execution.adverse_raw_price(
            self.position, bar.open, bar.high, bar.low
        )
        sl_slippage = self.slippage_bps if sl_touched else 0.0
        worst_intrabar_equity = self.guard.live_equity(
            self.position, adverse_price, slippage_bps=sl_slippage
        )
        breach = self.guard.check_breach(worst_intrabar_equity)
        if breach is not None:
            self._force_liquidate_on_breach(
                raw_price=adverse_price,
                exit_time=bar.timestamp,
                breach_status=breach,
                reason_text=f"Intrabar {breach.value} at raw price {adverse_price}",
                slippage_bps=sl_slippage,
            )
            return self._record_snapshot(
                timestamp=bar.timestamp,
                worst_intrabar_equity=worst_intrabar_equity,
                best_intrabar_equity=worst_intrabar_equity,
                close_equity=self.guard.closed_balance,
            )

        if sl_touched:
            self.execution.close_position(
                guard=self.guard,
                ledger=self.ledger,
                trades=self.trades,
                position=self.position,
                raw_price=adverse_price,
                exit_time=bar.timestamp,
                reason=ExitReason.STOP_LOSS,
                slippage_bps=self.slippage_bps,
            )
            self.position = None
            close_equity = self.guard.closed_balance
            if self.guard.check_target_reached(close_equity):
                self.guard.transition_status(AccountStatus.PASSED)
            return self._record_snapshot(
                timestamp=bar.timestamp,
                worst_intrabar_equity=worst_intrabar_equity,
                best_intrabar_equity=close_equity,
                close_equity=close_equity,
            )

        # Step C: Intrabar favorable path second
        best_raw_price, tp_touched = self.execution.favorable_raw_price(
            self.position, bar.open, bar.high, bar.low
        )
        best_intrabar_equity = self.guard.live_equity(self.position, best_raw_price)
        if self.guard.check_target_reached(best_intrabar_equity):
            open_equity = self.guard.live_equity(self.position, bar.open)
            if open_equity >= self.guard.profit_target_equity:
                target_raw = bar.open
            else:
                solved = self.execution.solve_target_raw_price(
                    self.position, self.guard, slippage_bps=0.0
                )
                target_raw = min(bar.high, max(bar.low, solved))
                if (
                    self.guard.live_equity(self.position, target_raw)
                    < self.guard.profit_target_equity - 1e-9
                ):
                    target_raw = best_raw_price
            self.execution.close_position(
                guard=self.guard,
                ledger=self.ledger,
                trades=self.trades,
                position=self.position,
                raw_price=target_raw,
                exit_time=bar.timestamp,
                reason=ExitReason.TARGET_AUTO_LIQUIDATE,
                slippage_bps=0.0,
            )
            self.position = None
            self.guard.transition_status(AccountStatus.PASSED)
            return self._record_snapshot(
                timestamp=bar.timestamp,
                worst_intrabar_equity=worst_intrabar_equity,
                best_intrabar_equity=best_intrabar_equity,
                close_equity=self.guard.closed_balance,
            )

        if tp_touched and self.position.take_profit is not None:
            self.execution.close_position(
                guard=self.guard,
                ledger=self.ledger,
                trades=self.trades,
                position=self.position,
                raw_price=self.position.take_profit,
                exit_time=bar.timestamp,
                reason=ExitReason.TAKE_PROFIT,
                slippage_bps=0.0,
            )
            self.position = None
            close_equity = self.guard.closed_balance
            return self._record_snapshot(
                timestamp=bar.timestamp,
                worst_intrabar_equity=worst_intrabar_equity,
                best_intrabar_equity=best_intrabar_equity,
                close_equity=close_equity,
            )

        close_equity = self.guard.live_equity(self.position, bar.close)
        return self._record_snapshot(
            timestamp=bar.timestamp,
            worst_intrabar_equity=worst_intrabar_equity,
            best_intrabar_equity=best_intrabar_equity,
            close_equity=close_equity,
        )

    def run(self, bars: Iterable[OhlcBar]) -> SimulationResult:
        """Process a stream of bars sequentially until exhausted or halted."""
        for bar in bars:
            if self.guard.status != AccountStatus.ACTIVE:
                break
            self.process_bar(bar)
        return self.get_result()

    def get_result(self) -> SimulationResult:
        """Return the immutable SimulationResult summarizing the run."""
        final_equity = (
            self.snapshots[-1].close_equity
            if self.snapshots
            else self.guard.closed_balance
        )
        return SimulationResult(
            status=self.guard.status,
            final_balance=self.guard.closed_balance,
            final_equity=final_equity,
            breach_timestamp=self.breach_timestamp,
            breach_reason=self.breach_reason,
            trades=list(self.trades),
            ledger=list(self.ledger),
            snapshots=list(self.snapshots),
        )


def run_strategy_simulation(
    bars: Sequence[OhlcBar],
    strategy: BaseStrategy,
    config: ChallengeConfig,
    slippage_bps: float = 0.0,
) -> SimulationResult:
    """Run a strategy sequentially across bars with strict next-bar execution."""
    sim = Simulator(config=config, slippage_bps=slippage_bps)
    pending_signal: Optional[StrategySignal] = None

    for index, bar in enumerate(bars):
        if sim.guard.status != AccountStatus.ACTIVE:
            break
        snapshot = sim.process_bar(bar, pending_signal=pending_signal)
        pending_signal = None
        if snapshot is None or sim.guard.status != AccountStatus.ACTIVE:
            break
        history_view = _BoundedHistoryView(bars, index + 1)
        pending_signal = strategy.on_bar(
            index=index,
            bar=bar,
            history_view=history_view,
            open_position=sim.position,
            equity=snapshot.close_equity,
        )
    return sim.get_result()
```

```python
from datetime import datetime, timedelta
from typing import Optional

from src.schemas.orders import ensure_utc_timestamp

ROLLOVER_INTERVAL_HOURS = 4


def check_4h_rollovers(prev_time: datetime, curr_time: datetime) -> list[datetime]:
    """Return all UTC 4-hour rollover boundaries strictly inside (prev_time, curr_time]."""
    prev_utc = ensure_utc_timestamp(prev_time)
    curr_utc = ensure_utc_timestamp(curr_time)
    if curr_utc <= prev_utc:
        return []

    base_hour = (prev_utc.hour // ROLLOVER_INTERVAL_HOURS) * ROLLOVER_INTERVAL_HOURS
    candidate = prev_utc.replace(
        hour=base_hour, minute=0, second=0, microsecond=0
    )
    if candidate <= prev_utc:
        candidate += timedelta(hours=ROLLOVER_INTERVAL_HOURS)

    rollovers: list[datetime] = []
    while candidate <= curr_utc:
        rollovers.append(candidate)
        candidate += timedelta(hours=ROLLOVER_INTERVAL_HOURS)
    return rollovers


def next_daily_reset_after(
    prev_time: datetime,
    reset_hour: int = 0,
    reset_minute: int = 30,
) -> datetime:
    """Return the first UTC daily reset timestamp strictly after prev_time."""
    prev_utc = ensure_utc_timestamp(prev_time)
    candidate = prev_utc.replace(
        hour=reset_hour, minute=reset_minute, second=0, microsecond=0
    )
    if candidate <= prev_utc:
        candidate += timedelta(days=1)
    return candidate


def crossed_daily_reset(
    prev_time: datetime,
    curr_time: datetime,
    reset_hour: int = 0,
    reset_minute: int = 30,
) -> bool:
    """Return True if (prev_time, curr_time] crosses the UTC daily reset boundary."""
    prev_utc = ensure_utc_timestamp(prev_time)
    curr_utc = ensure_utc_timestamp(curr_time)
    if curr_utc <= prev_utc:
        return False

    candidate = next_daily_reset_after(
        prev_utc, reset_hour=reset_hour, reset_minute=reset_minute
    )
    return candidate <= curr_utc


class EngineClock:
    """Sequential UTC clock tracking 4-hour swap boundaries and daily reset triggers."""

    def __init__(self, reset_hour: int = 0, reset_minute: int = 30) -> None:
        self.reset_hour = reset_hour
        self.reset_minute = reset_minute
        self.last_processed_time: Optional[datetime] = None

    def advance(self, curr_time: datetime) -> tuple[list[datetime], bool]:
        """Advance clock to curr_time and return (4h_rollovers, crossed_daily_reset)."""
        curr_utc = ensure_utc_timestamp(curr_time)
        if self.last_processed_time is None:
            self.last_processed_time = curr_utc
            return [], False
        if curr_utc < self.last_processed_time:
            raise ValueError("EngineClock received out-of-order timestamp.")

        rollovers = check_4h_rollovers(self.last_processed_time, curr_utc)
        did_reset = crossed_daily_reset(
            self.last_processed_time,
            curr_utc,
            reset_hour=self.reset_hour,
            reset_minute=self.reset_minute,
        )
        self.last_processed_time = curr_utc
        return rollovers, did_reset
```

```python
def _sample_trade_indices(
    rng: np.random.Generator,
    n_trades: int,
    max_trades_per_run: int,
    block_size: int,
) -> np.ndarray:
    if block_size <= 1:
        return rng.integers(0, n_trades, size=max_trades_per_run)
    n_blocks = (max_trades_per_run + block_size - 1) // block_size
    starts = rng.integers(0, n_trades, size=n_blocks)
    offsets = np.arange(block_size)
    indices = (starts[:, None] + offsets[None, :]) % n_trades
    return indices.ravel()[:max_trades_per_run]


def run_trade_bootstrap(
    trades: Sequence[TradeRecord],
    config: ChallengeConfig,
    n_simulations: int = 10_000,
    seed: int = 42,
    max_trades_per_run: int = 500,
    block_size: int = 1,
) -> MonteCarloReport:
    """Run seeded Monte Carlo bootstrap over historical TradeRecords through RulesGuard."""
    if not trades:
        raise ValueError("Cannot run Monte Carlo bootstrap on an empty trade sequence.")
    if n_simulations <= 0:
        raise ValueError("n_simulations must be positive.")
    if max_trades_per_run <= 0:
        raise ValueError("max_trades_per_run must be positive.")

    rng = np.random.default_rng(seed)
    n_trades = len(trades)

    # Infer typical time advance per trade so daily 00:30 UTC resets occur realistically.
    durations: list[timedelta] = []
    for i, trade in enumerate(trades):
        if i > 0 and trades[i].exit_time > trades[i - 1].exit_time:
            durations.append(trades[i].exit_time - trades[i - 1].exit_time)
        elif trade.exit_time > trade.entry_time:
            durations.append(trade.exit_time - trade.entry_time)
        else:
            durations.append(timedelta(hours=6))

    pass_count = 0
    mdl_count = 0
    mdd_count = 0
    truncated_count = 0
    trades_to_pass: list[int] = []
    final_balances = np.empty(n_simulations, dtype=float)

    base_start = datetime(2026, 1, 1, 1, 0, tzinfo=timezone.utc)

    for sim_idx in range(n_simulations):
        guard = RulesGuard(config)
        curr_time = base_start
        sampled_indices = _sample_trade_indices(
            rng, n_trades, max_trades_per_run, block_size
        )

        for step_num, t_idx in enumerate(sampled_indices, start=1):
            trade = trades[int(t_idx)]
            next_time = curr_time + durations[int(t_idx)]
            if crossed_daily_reset(
                curr_time,
                next_time,
                reset_hour=config.daily_reset_utc_hour,
                reset_minute=config.daily_reset_utc_minute,
            ):
                guard.on_daily_reset(current_close_equity=guard.closed_balance)
            curr_time = next_time

            projected_balance = guard.closed_balance + trade.net_pnl
            breach = guard.check_breach(projected_balance)
            if breach is not None:
                guard.record_balance_delta(trade.net_pnl, update_hwm=False)
                guard.transition_status(breach)
                break

            guard.record_balance_delta(trade.net_pnl, update_hwm=True)
            if guard.check_target_reached(guard.closed_balance):
                guard.closed_balance = min(
                    guard.closed_balance, guard.profit_target_equity
                )
                guard.transition_status(AccountStatus.PASSED)
                trades_to_pass.append(step_num)
                break

        if guard.status == AccountStatus.PASSED:
            pass_count += 1
        elif guard.status == AccountStatus.BREACHED_MDL:
            mdl_count += 1
        elif guard.status == AccountStatus.BREACHED_MDD:
            mdd_count += 1
        else:
            truncated_count += 1
        final_balances[sim_idx] = guard.closed_balance

    p10, p50, p90 = np.percentile(final_balances, [10, 50, 90])
    median_pass = float(np.median(trades_to_pass)) if trades_to_pass else None

    return MonteCarloReport(
        preset_name=config.name,
        n_simulations=n_simulations,
        seed=seed,
        pass_probability_pct=(pass_count / n_simulations) * 100.0,
        breach_mdl_pct=(mdl_count / n_simulations) * 100.0,
        breach_mdd_pct=(mdd_count / n_simulations) * 100.0,
        truncated_pct=(truncated_count / n_simulations) * 100.0,
        median_trades_to_pass=median_pass,
        p10_final_balance=float(p10),
        p50_final_balance=float(p50),
        p90_final_balance=float(p90),
    )


def run_friction_sensitivity(
    bars: Sequence[OhlcBar],
    strategy_factory: Callable[[], BaseStrategy],
    config: ChallengeConfig,
    slippage_bps_grid: Sequence[float] = (0.0, 1.0, 2.0, 4.0, 6.0, 8.0),
) -> list[FrictionSensitivityPoint]:
    """Evaluate strategy across a slippage grid to quantify execution drag."""
    points: list[FrictionSensitivityPoint] = []
    for slip_bps in slippage_bps_grid:
        strategy = strategy_factory()
        res = run_strategy_simulation(
            bars=bars, strategy=strategy, config=config, slippage_bps=float(slip_bps)
        )
        total_fees = sum(
            t.entry_fee + t.exit_fee + t.swap_fees for t in res.trades
        )
        points.append(
            FrictionSensitivityPoint(
                slippage_bps=float(slip_bps),
                status=res.status,
                final_balance=res.final_balance,
                net_pnl=res.final_balance - config.initial_balance,
                total_fees_usd=total_fees,
                trade_count=len(res.trades),
            )
        )
    return points
```

```python
from typing import Optional

from pydantic import BaseModel, ConfigDict, Field

from src.schemas.config import ChallengeConfig


class EvInputs(BaseModel):
    """Inputs required to compute the Expected Value ($ EV) of a prop challenge attempt."""

    model_config = ConfigDict(frozen=True, extra="forbid")

    entry_fee_usd: float = Field(ge=0)
    profit_target_usd: float = Field(gt=0)
    profit_split_pct: float = Field(gt=0, le=1.0)
    pass_probability: float = Field(ge=0.0, le=1.0)
    p_first_payout_given_pass: float = Field(default=0.70, gt=0.0, le=1.0)
    avg_gross_payout_usd: Optional[float] = Field(default=None, gt=0)
    avg_commissions_usd: float = Field(default=0.0, ge=0)
    avg_swap_fees_usd: float = Field(default=0.0, ge=0)


class ChallengeEvReport(BaseModel):
    """Expected Value ($ EV) and breakeven pass-rate summary for a challenge tier."""

    model_config = ConfigDict(frozen=True, extra="forbid")

    expected_gross_payout_usd: float
    expected_trader_payout_usd: float
    effective_payout_probability: float
    challenge_ev_usd: float
    breakeven_pass_probability: float
    total_fee_drag_usd: float


def calculate_challenge_ev(inputs: EvInputs) -> ChallengeEvReport:
    """Compute challenge Expected Value ($ EV), breakeven pass probability, and fee drag."""
    gross_payout = (
        inputs.avg_gross_payout_usd
        if inputs.avg_gross_payout_usd is not None
        else inputs.profit_target_usd
    )
    trader_payout = gross_payout * inputs.profit_split_pct
    effective_prob = inputs.pass_probability * inputs.p_first_payout_given_pass
    ev_usd = (effective_prob * trader_payout) - inputs.entry_fee_usd

    denom = inputs.p_first_payout_given_pass * trader_payout
    breakeven_pass_prob = inputs.entry_fee_usd / denom if denom > 0 else 0.0
    total_fee_drag = inputs.avg_commissions_usd + inputs.avg_swap_fees_usd

    return ChallengeEvReport(
        expected_gross_payout_usd=gross_payout,
        expected_trader_payout_usd=trader_payout,
        effective_payout_probability=effective_prob,
        challenge_ev_usd=ev_usd,
        breakeven_pass_probability=breakeven_pass_prob,
        total_fee_drag_usd=total_fee_drag,
    )


def calculate_ev_from_config(
    config: ChallengeConfig,
    pass_probability: float,
    p_first_payout_given_pass: float = 0.70,
    avg_gross_payout_usd: Optional[float] = None,
    avg_commissions_usd: float = 0.0,
    avg_swap_fees_usd: float = 0.0,
) -> ChallengeEvReport:
    """Convenience wrapper building EvInputs directly from a ChallengeConfig."""
    profit_target_usd = config.initial_balance * config.profit_target_pct
    return calculate_challenge_ev(
        EvInputs(
            entry_fee_usd=config.entry_fee_usd,
            profit_target_usd=profit_target_usd,
            profit_split_pct=config.profit_split_pct,
            pass_probability=pass_probability,
            p_first_payout_given_pass=p_first_payout_given_pass,
            avg_gross_payout_usd=avg_gross_payout_usd,
            avg_commissions_usd=avg_commissions_usd,
            avg_swap_fees_usd=avg_swap_fees_usd,
        )
    )
```

## src/strategies/session_sweep.py

```python
from typing import Optional, Sequence

from pydantic import BaseModel, ConfigDict, Field

from src.schemas.orders import OrderType, Position, Side
from src.schemas.telemetry import OhlcBar
from src.strategies.base import BaseStrategy, OrderIntent, StrategySignal


class SessionSweepConfig(BaseModel):
    """Configuration parameters for the Session Liquidity Sweep strategy."""

    model_config = ConfigDict(frozen=True, extra="forbid")

    symbol: str = "BTC/USD"
    lookback_bars: int = Field(default=20, ge=2)
    risk_per_trade_pct: float = Field(default=0.005, gt=0, le=0.05)
    stop_buffer_pct: float = Field(default=0.002, gt=0)
    reward_risk_ratio: float = Field(default=2.0, ge=1.0)
    max_leverage: float = Field(default=5.0, gt=0)


class SessionSweepStrategy(BaseStrategy):
    """Reference liquidity-sweep strategy trading false breakouts of prior range highs/lows."""

    def __init__(self, config: Optional[SessionSweepConfig] = None) -> None:
        self.config = config or SessionSweepConfig()

    def on_bar(
        self,
        index: int,
        bar: OhlcBar,
        history_view: Sequence[OhlcBar],
        open_position: Optional[Position],
        equity: float,
    ) -> Optional[StrategySignal]:
        if open_position is not None or equity <= 0:
            return None
        if (
            index < self.config.lookback_bars
            or len(history_view) <= self.config.lookback_bars
        ):
            return None

        prior_window = history_view[index - self.config.lookback_bars : index]
        prior_high = max(b.high for b in prior_window)
        prior_low = min(b.low for b in prior_window)

        # Bullish liquidity sweep: wick sweeps below prior_low and closes back above prior_low
        if bar.low < prior_low and bar.close > prior_low:
            stop_loss = min(bar.low, prior_low) * (1.0 - self.config.stop_buffer_pct)
            risk_per_unit = bar.close - stop_loss
            if risk_per_unit <= 0:
                return None
            take_profit = bar.close + self.config.reward_risk_ratio * risk_per_unit
            qty = self._compute_qty(equity, bar.close, risk_per_unit)
            if qty <= 0:
                return None
            return OrderIntent(
                symbol=self.config.symbol,
                side=Side.LONG,
                qty=qty,
                order_type=OrderType.MARKET,
                stop_loss=stop_loss,
                take_profit=take_profit,
            )

        # Bearish liquidity sweep: wick sweeps above prior_high and closes back below prior_high
        if bar.high > prior_high and bar.close < prior_high:
            stop_loss = max(bar.high, prior_high) * (1.0 + self.config.stop_buffer_pct)
            risk_per_unit = stop_loss - bar.close
            if risk_per_unit <= 0:
                return None
            take_profit = bar.close - self.config.reward_risk_ratio * risk_per_unit
            if take_profit <= 0:
                return None
            qty = self._compute_qty(equity, bar.close, risk_per_unit)
            if qty <= 0:
                return None
            return OrderIntent(
                symbol=self.config.symbol,
                side=Side.SHORT,
                qty=qty,
                order_type=OrderType.MARKET,
                stop_loss=stop_loss,
                take_profit=take_profit,
            )

        return None

    def _compute_qty(
        self, equity: float, reference_price: float, risk_per_unit: float
    ) -> float:
        risk_usd = equity * self.config.risk_per_trade_pct
        raw_qty = risk_usd / risk_per_unit
        max_notional_qty = (equity * self.config.max_leverage) / reference_price
        return max(0.0, min(raw_qty, max_notional_qty))
```
